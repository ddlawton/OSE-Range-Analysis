# Helper function to save gt tables as PNG and CSV
save_gt_png <- function(gt_obj, filename, subfolder = "locust_damage_treatment_region") {
  # Create output directories
  png_dir <- here::here("outputs", "tables", subfolder, "png")
  csv_dir <- here::here("outputs", "tables", subfolder, "csv")
  dir.create(png_dir, recursive = TRUE, showWarnings = FALSE)
  dir.create(csv_dir, recursive = TRUE, showWarnings = FALSE)
  
  # Save PNG
  png_path <- file.path(png_dir, filename)
  gtsave(gt_obj, png_path, vwidth = 3000)
  
  # Save CSV - extract the underlying data from the gt object
  csv_filename <- sub("\\.png$", ".csv", filename)
  csv_path <- file.path(csv_dir, csv_filename)
  
  # Extract data from gt object
  gt_data <- gt_obj$`_data`
  write_csv(gt_data, csv_path)
  
  return(invisible(list(png = png_path, csv = csv_path)))
}

# Table: annual OSE damage by region and field type

The data discrepancies seen the the OSE count x ground cover are really shown in this table as well. We can see that the trends are the same (e.g. fertilized plots have less than control), but the actual averages are totally different.

```{r}
#| echo: false
#| message: false
#| warning: false
#| 
ose_dat <- data |>
  select(region, fertilizer_treatment, ose_damage_percent) |>
  mutate(region = factor(region, levels = c('Saint Louis', 'Thies', 'Fatick', 'Kaffrine')),
         fertilizer_treatment = factor(fertilizer_treatment, levels = c('fertilized', 'control'))) |>
  drop_na(ose_damage_percent, fertilizer_treatment) |>
  group_by(region, fertilizer_treatment) |>
  summarize(ose_mean = mean(ose_damage_percent, na.rm = TRUE), .groups = "drop") |>
  pivot_wider(names_from = region, values_from = ose_mean)

# Create the gt table
tbl_damage_summary <- ose_dat |>
  gt(rowname_col = "fertilizer_treatment") |>
  tab_header(
    title = "Mean OSE damage by Region and Fertilizer Treatment"
  ) |>
  fmt_number(
    columns = everything(),
    decimals = 4
  ) |>
  cols_label(
    fertilizer_treatment = "Treatment",
    `Saint Louis` = "Saint Louis",
    Thies = "Thies",
    Fatick = "Fatick",
    Kaffrine = "Kaffrine"
  ) |>
  tab_style(
    style = cell_text(weight = "bold"),
    locations = cells_column_labels(everything())
  )

save_gt_png(tbl_damage_summary, "mean_ose_damage_by_region_treatment.png")
tbl_damage_summary


```
