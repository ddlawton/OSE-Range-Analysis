## Mean Annual Precipitation (MAP)

Mean annual precipitation (MAP) for Senegal was calculated using the CHIRPS dataset, which provides high-resolution (approximately 5 km) gridded precipitation estimates. We extracted CHIRPS daily precipitation data for Senegal for the most recent 10-year period (2015–2024). The data were spatially clipped to the national boundary, summed annually, and then averaged across years to produce a single raster layer representing mean annual precipitation.

```{r map-setup}
#| echo: false
#| message: false
#| warning: false

library(terra)
library(tidyterra)
library(sf)
library(rnaturalearth)
library(scales)

# Import the MAP TIFF
map_raster <- rast("data/processed/maps/senegal_map_2015_2024_chirps.tif")

# Get Senegal boundary and regions
senegal <- ne_states(country = "Senegal", returnclass = "sf")

# Define study regions
regions_of_interest <- c("Saint-Louis", "Thiès", "Fatick", "Kaffrine")
roi_sf <- senegal |> filter(name %in% regions_of_interest)
other_sf <- senegal |> filter(!name %in% regions_of_interest)

# Crop and mask raster to Senegal boundary
map_raster_senegal <- crop(map_raster, vect(senegal)) |>
  mask(vect(senegal))

# Create union of study regions for masking
roi_union <- st_union(roi_sf)
map_raster_roi <- mask(map_raster_senegal, vect(roi_union))
```

```{r}
library(terra)
library(sf)
library(smoothr)
library(ggplot2)
library(patchwork)
library(rlang)  # for !!sym() in aes

# Define months and raster file paths
months <- c("july", "august", "september", "october")
raster_paths <- paste0("data/processed/maps/senegal_precipitation_2021_", 
                       months, "_chirps.tif")

# Calculate global min/max across all months for consistent color scale
rainfall_values <- lapply(raster_paths, function(path) {
  r <- rast(path)
  values(r)[!is.na(values(r))]
})
all_values <- unlist(rainfall_values)
rainfall_min <- min(all_values)
rainfall_max <- max(all_values)

# --- Smooth the outlines ONCE outside loop, for speed ---
roi_sf_smooth      <- smoothr::smooth(roi_sf, method = "ksmooth", smoothness = 2)
senegal_sf_smooth  <- smoothr::smooth(senegal, method = "ksmooth", smoothness = 2)

# Helper function for full Senegal plot
plot_month_full <- function(month, raster_path, senegal, roi_sf,
                            senegal_sf_smooth, roi_sf_smooth, min_val, max_val) {
  raster <- rast(raster_path)
  # 1. Upsample (smooth raster-edged look)
  raster <- terra::disagg(raster, fact = 4)
  # 2. Mask to Senegal boundary
  senegal_vect <- vect(senegal)
  raster_cropped <- crop(raster, senegal_vect)
  raster_masked  <- mask(raster_cropped, senegal_vect)

  # -- Layered ggplot: smoothed outlines, no ugly squares, constant scale! --
  ggplot() +
    stars::geom_stars(data = stars::st_as_stars(raster_masked)) +
    geom_sf(data = senegal_sf_smooth, fill = NA, color = "grey80", linewidth = 1.5) +
    geom_sf(data = roi_sf_smooth, fill = NA, color = 'gray85', linewidth = 2) +
    geom_sf(data = roi_sf_smooth, fill = NA, color = 'black', linewidth = 1) +
    scale_fill_viridis_c(
      name = "Rainfall (mm)",
      na.value = NA,
      limits = c(min_val, max_val), 
      option = "D", 
      direction = -1
    ) +
    theme_minimal() +
    labs(title = paste(tools::toTitleCase(month), "2021")) +
    theme(axis.text = element_blank(), 
          axis.ticks = element_blank(), 
          panel.grid = element_blank()) +
    xlab('') +
    ylab('')
}

# Create plots
plots_full <- mapply(
  plot_month_full, months, raster_paths,
  MoreArgs = list(
    senegal = senegal, 
    roi_sf = roi_sf, 
    senegal_sf_smooth = senegal_sf_smooth, 
    roi_sf_smooth = roi_sf_smooth, 
    min_val = rainfall_min, 
    max_val = rainfall_max
  ),
  SIMPLIFY = FALSE
)

# Combine in 2x2 grid, grouped legend
(final_plot <- (plots_full[[1]] | plots_full[[2]]) /
  (plots_full[[3]] | plots_full[[4]]) + 
  patchwork::plot_layout(guides = "collect") & 
  theme(legend.position = "bottom"))
```

```{r}
library(terra)
library(sf)
library(dplyr)

# Ensure roi_sf is in the same CRS as the rasters
roi_sf <- st_transform(roi_sf, crs(rast(raster_paths[1])))
# Function to extract mean and sd rainfall for each region in roi_sf for a given raster
extract_monthly_stats <- function(raster_path, polygons, month) {
  r <- rast(raster_path)
  # Extract mean
  mean_df <- terra::extract(r, vect(polygons), fun = mean, na.rm = TRUE)
  # Extract sd
  sd_df <- terra::extract(r, vect(polygons), fun = sd, na.rm = TRUE)
  # Combine and add region and month
  out <- mean_df |>
    rename(mean_rainfall = 2) |>
    left_join(sd_df |> rename(sd_rainfall = 2), by = "ID") |>
    mutate(region = polygons$name, month = month)
  out
}

# Loop over months and get stats
rainfall_stats <- lapply(seq_along(months), function(i) {
  extract_monthly_stats(raster_paths[i], roi_sf, months[i])
})

# Combine into one data frame
rainfall_stats_df <- bind_rows(rainfall_stats)

# Clean up columns if needed
rainfall_stats_df <- rainfall_stats_df |>
  select(region, month, mean_rainfall, sd_rainfall)


rainfall_stats_df |>
  as_tibble() |>
  drop_na(mean_rainfall) |>
  select(region, month, mean_rainfall, sd_rainfall) |>
  mutate(
    mean_rainfall = round(mean_rainfall, 1),
    sd_rainfall = round(sd_rainfall, 1)
  ) |>
  gt(rowname_col = "region") |>
  tab_header(
    title = "Average Monthly Rainfall by Region (2021)"
  ) |>
  cols_label(
    region = "Region",
    month = "Month",
    mean_rainfall = "Mean Rainfall (mm)",
    sd_rainfall = "SD (mm)"
  ) |>
  fmt_number(
    columns = c(mean_rainfall, sd_rainfall),
    decimals = 1
  ) |>
  tab_options(
    table.font.size = "small"
  )

```
