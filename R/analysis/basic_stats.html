# Basic statistics around dataset

```{r basic-statistics}
#| echo: false
#| message: false
#| warning: false

summary_table <- table_variable_summary(data)

# Also save the detailed variable summary
if (exists("summary_table")) {
  # Extract the underlying data if it's a gt table
  tryCatch({
    if ("gt_tbl" %in% class(summary_table)) {
      # For gt tables, we need to recreate the summary data
      summary_data <- data |>
        select(where(is.numeric)) |>
        summarise(across(everything(), list(
          min = ~min(.x, na.rm = TRUE),
          max = ~max(.x, na.rm = TRUE), 
          mean = ~mean(.x, na.rm = TRUE),
          median = ~median(.x, na.rm = TRUE),
          sd = ~sd(.x, na.rm = TRUE)
        ))) |>
        pivot_longer(everything(), names_to = "variable_stat", values_to = "value") |>
        separate(variable_stat, into = c("variable", "statistic"), sep = "_(?=[^_]*$)")
      
      save_data_table(summary_data, "variable_summary_statistics")
    }
  }, error = function(e) {
    message("Could not save summary table data: ", e$message)
  })
}

# Display the summary table
summary_table
```
